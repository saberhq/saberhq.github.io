{{- /* The page's preview image: its own picture, else its cover art, else the About photo. Returns url, width, height, alt. */ -}} {{- $r := dict -}} {{- with .Params.images -}} {{- with $.Resources.Get (index . 0) -}} {{- $r = dict "url" .Permalink "width" .Width "height" .Height "alt" ($.Params.photoAlt | default "") -}} {{- end -}} {{- end -}} {{- if and (not $r) .Params.image -}} {{- $path := .Params.image -}} {{- $r = dict "url" (absURL $path) "alt" (printf "Cover art for %s" .Title) -}} {{- with try (images.Config (path.Join "static" $path)) -}} {{- with .Value }}{{ $r = merge $r (dict "width" .Width "height" .Height) }}{{ end -}} {{- end -}} {{- end -}} {{- if not $r -}} {{- with site.GetPage "/about" -}} {{- with .Resources.Get (.Params.photo | default "profile.jpg") -}} {{- $r = dict "url" .Permalink "width" .Width "height" .Height "alt" (printf "Photo of %s" site.Params.author) -}} {{- end -}} {{- end -}} {{- end -}} {{- return $r -}}
